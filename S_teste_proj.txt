O importante é confirmarmos:

remote origin  → GitHub
working tree   → limpo
branch atual   → provavelmente main
commit         → identificado
origin/HEAD    → identificado

Se estiver limpo, crie sua branch:

git switch -c feature/manuel-local-validation

Escolhi feature/... porque é compatível com o padrão que já vimos nas configurações anteriores do projeto; depois vamos conferir o workflow atual do GitHub clonado antes de assumir qualquer regra definitiva.

section "14. GATEWAY OUTBOUND / mTLS"

###############################################################################
# Descobre URL HTTPS declarada pelo projeto.
###############################################################################

GATEWAY_URL="$(
    grep -Eo 'https://[^ "'\'']+' agent_config.yaml 2>/dev/null \
        | grep -E 'llm|agentes|outbound|api' \
        | head -1
)"

if [[ -z "$GATEWAY_URL" ]]; then
    echo "URL não extraída automaticamente de agent_config.yaml."
else
    echo "Gateway declarado:"
    echo "$GATEWAY_URL"
fi

CERT="${KEY_STORE_CERT_PATH:-}"
KEY="${KEY_STORE_KEY_PATH:-}"
CA="${TRUST_STORE_CA_PATH:-${REQUESTS_CA_BUNDLE:-}}"

MTLS_READY=1

[[ -f "$CERT" ]] || MTLS_READY=0
[[ -f "$KEY"  ]] || MTLS_READY=0
[[ -f "$CA"   ]] || MTLS_READY=0

if [[ -z "$GATEWAY_URL" ]]; then

    warn 14 "Não foi possível identificar automaticamente o Gateway."

elif (( MTLS_READY == 0 )); then

    warn 14 "Certificado, chave e/ou CA de mTLS ainda não estão disponíveis."

else

    echo
    echo "Testando negociação TLS com certificado cliente."
    echo "Nenhum segredo será exibido."

    MTLS_RESULT="$(
        curl \
            -sS \
            --connect-timeout 10 \
            --max-time 20 \
            --cert "$CERT" \
            --key "$KEY" \
            --cacert "$CA" \
            -o /dev/null \
            -w 'HTTP=%{http_code} | IP=%{remote_ip} | TLS=%{ssl_verify_result}' \
            "$GATEWAY_URL" \
            2>&1
    )"

    MTLS_RC=$?

    echo "$MTLS_RESULT"

    if (( MTLS_RC == 0 )); then
        pass 14 "Handshake HTTPS/mTLS concluído. O HTTP pode ser 4xx sem chamada de API válida."
    else
        fail 14 "Falha na negociação HTTPS/mTLS com o Gateway."
    fi
fi
