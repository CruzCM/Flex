echo "------------------------------------------------------------"
TMP="/tmp/ambiente-venv-$$"

    echo "venv: OK"
    "$TMP/bin/python" --version
    "$TMP/bin/python" -m pip --version 2>&1 || true
else

    fontes.intranet.bb.com.br \

    echo
    echo ">> https://$host"
    curl \
        -sS \
        --connect-timeout 8 \
        --max-time 15 \
        -o /dev/null \
        -w 'HTTP=%{http_code} | IP=%{remote_ip} | TLS=%{ssl_verify_result}\n' \
        "https://$host" || true
done
