%%spark

jvm = spark._jvm

context_loader = (
    jvm.java.lang.Thread
    .currentThread()
    .getContextClassLoader()
)

driver_class = jvm.java.lang.Class.forName(
    cliente_oracle.driver,
    True,
    context_loader,
)

driver_instance = driver_class.newInstance()

props = jvm.java.util.Properties()
props.setProperty("user", cliente_oracle.user)
props.setProperty("password", cliente_oracle.password)

conn = None
stmt = None
resultado = None

try:
    conn = driver_instance.connect(
        cliente_oracle.url,
        props,
    )

    if conn is None:
        raise RuntimeError(
            "O driver Oracle não reconheceu a URL JDBC."
        )

    stmt = conn.createStatement()
    resultado = stmt.executeQuery("SELECT 1 AS QTD FROM DUAL")

    if not resultado.next():
        raise RuntimeError("SELECT não retornou uma linha.")

    print("Driver:", driver_instance.getClass().getName())
    print("Conexão direta por driver.connect: OK")
    print("SELECT 1 FROM DUAL:", int(resultado.getInt(1)))

finally:
    if resultado is not None:
        resultado.close()

    if stmt is not None:
        stmt.close()

    if conn is not None:
        conn.close()
