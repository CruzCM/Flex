%%spark

jvm = spark._jvm
driver = cliente_oracle.driver

print("=" * 70)
print("1. JARS REGISTRADOS NA SESSÃO")
print("=" * 70)

spark_jars = spark.sparkContext.getConf().get("spark.jars", "")
print("spark.jars:", spark_jars or "<vazio>")

jars_contexto = []
try:
    iterator = spark._jsc.sc().listJars().iterator()
    while iterator.hasNext():
        jars_contexto.append(str(iterator.next()))
except Exception as exc:
    print("Falha ao consultar SparkContext.listJars():", exc)

print("SparkContext.listJars():")
for jar in jars_contexto:
    print(" -", jar)

print("SparkFiles root:", jvm.org.apache.spark.SparkFiles.getRootDirectory())

print()
print("=" * 70)
print("2. CLASSLOADER PADRÃO")
print("=" * 70)

classe_driver = None

try:
    classe_driver = jvm.java.lang.Class.forName(driver)
    print("Class.forName padrão: OK")
except Exception as exc:
    print("Class.forName padrão: FALHOU")
    print(type(exc).__name__, str(exc))

print()
print("=" * 70)
print("3. CONTEXT CLASSLOADER")
print("=" * 70)

try:
    context_loader = (
        jvm.java.lang.Thread
        .currentThread()
        .getContextClassLoader()
    )

    classe_driver = jvm.java.lang.Class.forName(
        driver,
        True,
        context_loader,
    )

    print("ContextClassLoader:", context_loader.getClass().getName())
    print("Carga pelo ContextClassLoader: OK")

except Exception as exc:
    print("Carga pelo ContextClassLoader: FALHOU")
    print(type(exc).__name__, str(exc))

print()
print("=" * 70)
print("4. CONEXÃO JDBC DIRETA — SOMENTE SELECT")
print("=" * 70)

conn = None
stmt = None
resultado = None

try:
    conn = jvm.java.sql.DriverManager.getConnection(
        cliente_oracle.url,
        cliente_oracle.user,
        cliente_oracle.password,
    )

    stmt = conn.createStatement()
    resultado = stmt.executeQuery("SELECT 1 AS QTD FROM DUAL")

    if not resultado.next():
        raise RuntimeError("SELECT de diagnóstico não retornou linha.")

    print("Conexão JDBC direta: OK")
    print("SELECT 1 FROM DUAL:", int(resultado.getInt(1)))

except Exception as exc:
    print("Conexão JDBC direta: FALHOU")
    print(type(exc).__name__, str(exc))

finally:
    if resultado is not None:
        try:
            resultado.close()
        except Exception:
            pass

    if stmt is not None:
        try:
            stmt.close()
        except Exception:
            pass

    if conn is not None:
        try:
            conn.close()
        except Exception:
            pass
