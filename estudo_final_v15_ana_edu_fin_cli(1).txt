def linha_div(char="=", tam=126):
    return char * tam


def titulo_secao(num, titulo):
    print("\n" + linha_div("="))
    print(f" [{num:02d}] {titulo.upper()}")
    print(linha_div("="))


def subtitulo(titulo):
    print("\n" + linha_div("-"))
    print(" " + titulo)
    print(linha_div("-"))


def format_num(val):
    if val is None:
        return "-"
    try:
        return f"{int(val):,}".replace(",", ".")
    except Exception:
        return str(val)


def format_moeda(val):
    if val is None:
        return "-"
    try:
        v = float(val)
        return f"R$ {v:,.2f}".replace(",", "X").replace(".", ",").replace("X", ".")
    except Exception:
        return str(val)


def format_pct(val):
    if val is None:
        return "-"
    try:
        return f"{float(val) * 100:.2f}%".replace(".", ",")
    except Exception:
        return str(val)


def format_decimal(val, casas=6):
    if val is None:
        return "-"
    try:
        return f"{float(val):.{casas}f}".replace(".", ",")
    except Exception:
        return str(val)


def exibir_tabela_formatada(colunas, linhas, alinhamentos=None):
    if not linhas:
        print("  (Nenhum dado encontrado)")
        return

    linhas_txt = []
    for lin in linhas:
        linhas_txt.append(["-" if v is None else str(v) for v in lin])

    larguras = [len(str(c)) for c in colunas]
    for lin in linhas_txt:
        for i, val in enumerate(lin):
            larguras[i] = max(larguras[i], len(val))

    if alinhamentos is None:
        alinhamentos = ["E"] * len(colunas)

    header = " | ".join(
        f"{str(c):<{larguras[i]}}" if alinhamentos[i] == "E"
        else f"{str(c):>{larguras[i]}}"
        for i, c in enumerate(colunas)
    )
    print("  " + header)
    print("  " + "-+-".join("-" * w for w in larguras))

    for lin in linhas_txt:
        texto = " | ".join(
            f"{lin[i]:<{larguras[i]}}" if alinhamentos[i] == "E"
            else f"{lin[i]:>{larguras[i]}}"
            for i in range(len(colunas))
        )
        print("  " + texto)


def imprimir_dict_vertical(titulo, dados, ordem=None):
    subtitulo(titulo)
    if ordem is None:
        ordem = list(dados.keys())
    largura = max(len(str(k)) for k in ordem) if ordem else 1
    for k in ordem:
        print(f"  {k:<{largura}} : {dados.get(k)}")

geral = spark.sql(f"""
SELECT
    COUNT(*) AS TOTAL_REGISTROS,
    COUNT(DISTINCT CD_CLI) AS CLIENTES_DISTINTOS,
    COUNT(*) - COUNT(DISTINCT CD_CLI) AS DUPLICIDADES_CD_CLI,

    SUM(CASE WHEN CD_CLI IS NULL THEN 1 ELSE 0 END) AS CD_CLI_NULO,

    MIN(DT_EXEA) AS MIN_DT_EXEA,
    MAX(DT_EXEA) AS MAX_DT_EXEA,
    MIN(DT_MES_EXEA) AS MIN_DT_MES_EXEA,
    MAX(DT_MES_EXEA) AS MAX_DT_MES_EXEA,

    MIN(DT_REF_INI) AS MIN_DT_REF_INI,
    MAX(DT_REF_INI) AS MAX_DT_REF_INI,
    MIN(DT_REF_FIM) AS MIN_DT_REF_FIM,
    MAX(DT_REF_FIM) AS MAX_DT_REF_FIM,

    MIN(DATEDIFF(DT_REF_FIM, DT_REF_INI) + 1) AS MIN_DIAS_JANELA,
    MAX(DATEDIFF(DT_REF_FIM, DT_REF_INI) + 1) AS MAX_DIAS_JANELA,

    SUM(CASE WHEN DT_REF_INI IS NULL THEN 1 ELSE 0 END) AS DT_REF_INI_NULA,
    SUM(CASE WHEN DT_REF_FIM IS NULL THEN 1 ELSE 0 END) AS DT_REF_FIM_NULA,
    SUM(CASE WHEN DT_REF_INI > DT_REF_FIM THEN 1 ELSE 0 END) AS JANELA_INVERTIDA,

    SUM(CASE WHEN QT_TRANS_TOTAL = 0 THEN 1 ELSE 0 END) AS CLI_SEM_MOVIMENTO,
    SUM(CASE WHEN QT_TRANS_TOTAL > 0 THEN 1 ELSE 0 END) AS CLI_COM_MOVIMENTO,

    SUM(CASE
            WHEN QT_TRANS_TOTAL > 0
             AND QT_TRANS_TOTAL > COALESCE(QT_TRANS_ENT,0) + COALESCE(QT_TRANS_SAI,0)
            THEN 1 ELSE 0
        END) AS CLI_COM_NATUREZA_FORA_C_D,

    SUM(CASE
            WHEN QT_TRANS_TOTAL > 0
            THEN QT_TRANS_TOTAL - COALESCE(QT_TRANS_ENT,0) - COALESCE(QT_TRANS_SAI,0)
            ELSE 0
        END) AS QT_TRANS_NATUREZA_FORA_C_D

FROM {VIEW_BASE}
""").first()

renda_sentinelas = spark.sql(f"""
SELECT
    CASE
        WHEN VL_REN_PRES = -1.00 THEN '[-1] Sem escoragem Behaviour/Bureau'
        WHEN VL_REN_PRES = -2.00 THEN '[-2] CPF inválido'
        WHEN VL_REN_PRES = -3.00 THEN '[-3] Não PF'
        WHEN VL_REN_PRES = -4.00 THEN '[-4] Identidade ambígua'
        WHEN VL_REN_PRES = -5.00 THEN '[-5] Duplicidade no modelo'
        WHEN VL_REN_PRES = -6.00 THEN '[-6] Renda nula na origem'
        WHEN VL_REN_PRES = -7.00 THEN '[-7] Renda negativa na origem'
        WHEN VL_REN_PRES = 0.00  THEN '[0] Modelo válido, renda zero'
        WHEN VL_REN_PRES > 0.00  THEN '[>0] Renda positiva utilizável'
        ELSE '[OUTROS] Valor/código não previsto'
    END AS DIAGNOSTICO,
    COUNT(*) AS QTD,

FROM {VIEW_BASE}
GROUP BY 1
ORDER BY QTD DESC
""").collect()

spark.sql(f"""
CREATE OR REPLACE TEMP VIEW {VIEW_RECALC} AS

WITH r1 AS (
    SELECT
        b.*,

        CASE
            WHEN CD_MAC_PRFL_CLI IS NULL THEN NULL
            WHEN CD_MAC_PRFL_CLI = 1 THEN 'Endividado'
            WHEN CD_MAC_PRFL_CLI = 2 THEN 'Equilibrista'
            WHEN CD_MAC_PRFL_CLI = 3 THEN 'Investidor'
            ELSE 'CODIGO NAO MAPEADO'
        END AS EXP_NM_MAC_PRFL_CLI,

        CASE
            WHEN CD_MIC_PRFL_CLI IS NULL THEN NULL
            WHEN CD_MIC_PRFL_CLI = 1 THEN 'Inadimplente'
            WHEN CD_MIC_PRFL_CLI = 2 THEN 'Acrobata'
            WHEN CD_MIC_PRFL_CLI = 3 THEN 'Iminente'
            WHEN CD_MIC_PRFL_CLI = 4 THEN 'Consciente'
            WHEN CD_MIC_PRFL_CLI = 5 THEN 'Equilibrista'
            WHEN CD_MIC_PRFL_CLI = 6 THEN 'Acelerado'
            WHEN CD_MIC_PRFL_CLI = 7 THEN 'Precavido'
            WHEN CD_MIC_PRFL_CLI = 8 THEN 'Despreocupado'
            ELSE 'CODIGO NAO MAPEADO'
        END AS EXP_NM_MIC_PRFL_CLI,

        CAST(
              COALESCE(VL_ENT_REN, 0)
            + COALESCE(VL_ENT_EST, 0)
            + COALESCE(VL_ENT_RESG, 0)
            + COALESCE(VL_ENT_OUT, 0)
            + COALESCE(VL_ENT_CRED, 0)
            AS DECIMAL(18,2)
        ) AS EXP_VL_ENT_TOTAL,

        CAST(
              COALESCE(VL_SAI_IND, 0)
            + COALESCE(VL_SAI_ESS, 0)
            + COALESCE(VL_SAI_NAO_ESS, 0)
            + COALESCE(VL_SAI_FUT, 0)
            + COALESCE(VL_SAI_OBR, 0)
            AS DECIMAL(18,2)
        ) AS EXP_VL_SAI_TOTAL

    FROM {VIEW_BASE} b
),

r2 AS (
    SELECT
        r1.*,

        CASE
            WHEN EXP_NM_MAC_PRFL_CLI IS NULL OR EXP_NM_MIC_PRFL_CLI IS NULL THEN NULL
            WHEN EXP_NM_MAC_PRFL_CLI = 'CODIGO NAO MAPEADO'
              OR EXP_NM_MIC_PRFL_CLI = 'CODIGO NAO MAPEADO' THEN 'CODIGO NAO MAPEADO'
            WHEN EXP_NM_MAC_PRFL_CLI = 'Equilibrista'
             AND EXP_NM_MIC_PRFL_CLI = 'Equilibrista' THEN 'Equilibrista'
            ELSE CONCAT(EXP_NM_MAC_PRFL_CLI, ' ', EXP_NM_MIC_PRFL_CLI)
        END AS EXP_NM_PRFL_FIN,

        CAST(
            CASE
                WHEN QT_TRANS_TOTAL = 0 THEN NULL
                ELSE EXP_VL_ENT_TOTAL - EXP_VL_SAI_TOTAL
            END AS DECIMAL(18,2)
        ) AS EXP_VL_RES_ORC,

        CAST(
            COALESCE(EXP_VL_SAI_TOTAL / NULLIF(EXP_VL_ENT_TOTAL, 0), 0)
            AS DECIMAL(9,6)
        ) AS EXP_PC_SAI_ENT

    FROM r1
),

r3 AS (
    SELECT
        r2.*,

        CAST(
            CASE
                WHEN QT_TRANS_TOTAL = 0 THEN NULL
                WHEN EXP_PC_SAI_ENT BETWEEN 0.950000 AND 1.050000 THEN 0
                WHEN EXP_PC_SAI_ENT > 1.050000 AND EXP_PC_SAI_ENT <= 1.250000 THEN 1
                WHEN EXP_PC_SAI_ENT > 1.250000 THEN 2
                WHEN EXP_PC_SAI_ENT >= 0.750000 AND EXP_PC_SAI_ENT < 0.950000 THEN 3
                ELSE 4
            END AS INT
        ) AS EXP_CD_FAIXA_ORC,

        CAST(CASE WHEN VL_REN_PRES > 0 THEN COALESCE(VL_SAI_IND / VL_REN_PRES, 0) ELSE 0 END AS DECIMAL(9,6)) AS EXP_PC_SAI_IND,
        CAST(CASE WHEN VL_REN_PRES > 0 THEN COALESCE(VL_SAI_ESS / VL_REN_PRES, 0) ELSE 0 END AS DECIMAL(9,6)) AS EXP_PC_SAI_ESS,
        CAST(CASE WHEN VL_REN_PRES > 0 THEN COALESCE(VL_SAI_NAO_ESS / VL_REN_PRES, 0) ELSE 0 END AS DECIMAL(9,6)) AS EXP_PC_SAI_NAO_ESS,
        CAST(CASE WHEN VL_REN_PRES > 0 THEN COALESCE(VL_SAI_FUT / VL_REN_PRES, 0) ELSE 0 END AS DECIMAL(9,6)) AS EXP_PC_SAI_FUT,
        CAST(CASE WHEN VL_REN_PRES > 0 THEN COALESCE(VL_SAI_OBR / VL_REN_PRES, 0) ELSE 0 END AS DECIMAL(9,6)) AS EXP_PC_SAI_OBR

recon = spark.sql(f"""
SELECT
    COUNT(*) AS BASE,

    SUM(CASE WHEN NOT (NM_MAC_PRFL_CLI <=> EXP_NM_MAC_PRFL_CLI) THEN 1 ELSE 0 END) AS ERR_NM_MAC,
