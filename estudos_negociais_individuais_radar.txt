SELECT
    INTEGER(CD_MAC_PRFL_CLI) AS CD_MAC_PRFL_CLI,
    VARCHAR(TRIM(NM_MAC_PRFL_CLI), 100) AS NM_MAC_PRFL_CLI,
    INTEGER(CD_MIC_PRFL_CLI) AS CD_MIC_PRFL_CLI,

# 3. Como DT_REF funciona por cliente
#    Toda a redução ocorre no DB2; retorna apenas o resumo.
# --------------------------------------------------------------------------------------------------

sql_dt_ref_resumo = f"""
WITH POR_CLIENTE AS (
    SELECT
        CD_CLI,
        COUNT(*) AS QT_REGISTROS,
        COUNT(DISTINCT DT_REF) AS QT_DT_REF,
        MIN(DT_REF) AS DT_REF_MIN,
        MAX(DT_REF) AS DT_REF_MAX
    FROM DB2D1D.DVS_GRDR_FNCO_PF
    WHERE DT_REF >= DATE('{DATA_INI}')
      AND DT_REF <= DATE('{data_fim}')
    GROUP BY CD_CLI
)
SELECT
    BIGINT(COUNT(*)) AS QT_CLIENTES,
    BIGINT(SUM(CASE WHEN QT_REGISTROS = 1 THEN 1 ELSE 0 END)) AS CLI_1_REGISTRO,
    BIGINT(SUM(CASE WHEN QT_REGISTROS > 1 THEN 1 ELSE 0 END)) AS CLI_MAIS_1_REGISTRO,
    BIGINT(SUM(CASE WHEN QT_DT_REF = 1 THEN 1 ELSE 0 END)) AS CLI_1_DT_REF,
    BIGINT(SUM(CASE WHEN QT_DT_REF > 1 THEN 1 ELSE 0 END)) AS CLI_MAIS_1_DT_REF,
    BIGINT(MAX(QT_REGISTROS)) AS MAX_REGISTROS_CLIENTE,
    BIGINT(MAX(QT_DT_REF)) AS MAX_DT_REF_CLIENTE,
    MIN(DT_REF_MIN) AS MENOR_DT_REF,
    MAX(DT_REF_MAX) AS MAIOR_DT_REF
FROM POR_CLIENTE
"""

df_dt_ref_resumo = conector_db2.sql(
    sql_dt_ref_resumo,
    fetchsize=10_000,
    query_timeout=15 * 60,
)

print("\n3. COMO DT_REF SE COMPORTA POR CLIENTE")

# --------------------------------------------------------------------------------------------------
# 4. Distribuição da quantidade de DT_REF por cliente
# --------------------------------------------------------------------------------------------------

sql_dt_ref_dist = f"""
WITH POR_CLIENTE AS (
    SELECT
        CD_CLI,
        COUNT(DISTINCT DT_REF) AS QT_DT_REF
    FROM DB2D1D.DVS_GRDR_FNCO_PF
    WHERE DT_REF >= DATE('{DATA_INI}')
      AND DT_REF <= DATE('{data_fim}')
    GROUP BY CD_CLI
)
SELECT
    CASE
        WHEN QT_DT_REF = 1 THEN '01'
        WHEN QT_DT_REF = 2 THEN '02'
        WHEN QT_DT_REF = 3 THEN '03'
        WHEN QT_DT_REF BETWEEN 4 AND 6 THEN '04_A_06'
        WHEN QT_DT_REF BETWEEN 7 AND 12 THEN '07_A_12'
        ELSE '13_OU_MAIS'
    END AS FAIXA_QT_DT_REF,
    BIGINT(COUNT(*)) AS QT_CLIENTES
FROM POR_CLIENTE
GROUP BY
    CASE
        WHEN QT_DT_REF = 1 THEN '01'
        WHEN QT_DT_REF = 2 THEN '02'
        WHEN QT_DT_REF = 3 THEN '03'
        WHEN QT_DT_REF BETWEEN 4 AND 6 THEN '04_A_06'
        WHEN QT_DT_REF BETWEEN 7 AND 12 THEN '07_A_12'
        ELSE '13_OU_MAIS'
    END
ORDER BY FAIXA_QT_DT_REF
"""

print("\n4. DISTRIBUIÇÃO — QUANTAS DT_REF UM MESMO CLIENTE POSSUI")
conector_db2.sql(
    sql_dt_ref_dist,

).show(100, truncate=False)


# --------------------------------------------------------------------------------------------------
# 5. Empates na DT_REF máxima
#    Responde se MAX(DT_REF) sozinho basta ou se existe mais de uma linha na data mais recente.
# --------------------------------------------------------------------------------------------------

sql_empate_ultima_data = f"""
WITH P AS (
    SELECT
        DT_REF,
        CD_CLI,
        CD_MAC_PRFL_CLI,
        CD_MIC_PRFL_CLI
    FROM DB2D1D.DVS_GRDR_FNCO_PF
    WHERE DT_REF >= DATE('{DATA_INI}')
      AND DT_REF <= DATE('{data_fim}')
),
MAX_REF AS (
    SELECT
        CD_CLI,
        MAX(DT_REF) AS DT_REF_MAX
    FROM P
    GROUP BY CD_CLI
),
ULTIMA AS (
    SELECT
        P.CD_CLI,
        P.DT_REF,
        COUNT(*) AS QT_REG_ULTIMA_DATA,
        MIN(COALESCE(P.CD_MAC_PRFL_CLI, -999)) AS MAC_MIN,
        MAX(COALESCE(P.CD_MAC_PRFL_CLI, -999)) AS MAC_MAX,
        MIN(COALESCE(P.CD_MIC_PRFL_CLI, -999)) AS MIC_MIN,
        MAX(COALESCE(P.CD_MIC_PRFL_CLI, -999)) AS MIC_MAX
    FROM P
    INNER JOIN MAX_REF M
        ON P.CD_CLI = M.CD_CLI
       AND P.DT_REF = M.DT_REF_MAX
    GROUP BY
        P.CD_CLI,
        P.DT_REF
)
SELECT
    BIGINT(COUNT(*)) AS QT_CLIENTES,
    BIGINT(SUM(CASE WHEN QT_REG_ULTIMA_DATA = 1 THEN 1 ELSE 0 END)) AS ULTIMA_DATA_1_REG,
    BIGINT(SUM(CASE WHEN QT_REG_ULTIMA_DATA > 1 THEN 1 ELSE 0 END)) AS ULTIMA_DATA_MULT_REG,
    BIGINT(SUM(
        CASE
            WHEN QT_REG_ULTIMA_DATA > 1
             AND (MAC_MIN <> MAC_MAX OR MIC_MIN <> MIC_MAX)
            THEN 1 ELSE 0
        END
    )) AS ULTIMA_DATA_MULT_PERFIL_DIFERENTE,
    BIGINT(MAX(QT_REG_ULTIMA_DATA)) AS MAX_REG_NA_ULTIMA_DATA
FROM ULTIMA
"""

print("\n5. EMPATES NA DT_REF MAIS RECENTE")
conector_db2.sql(
    sql_empate_ultima_data,
    fetchsize=10_000,
    query_timeout=15 * 60,
).show(truncate=False)

# 6. O perfil que a regra ATUAL escolheria:

sql_escolhido = f"""
WITH ORDENADO AS (

        ) AS RN
    FROM DB2D1D.DVS_GRDR_FNCO_PF
    WHERE DT_REF >= DATE('{DATA_INI}')
      AND DT_REF <= DATE('{data_fim}')
)
SELECT
    INTEGER(CD_MAC_PRFL_CLI) AS CD_MAC_PRFL_CLI,
    VARCHAR(TRIM(NM_MAC_PRFL_CLI), 100) AS NM_MAC_PRFL_CLI,
    INTEGER(CD_MIC_PRFL_CLI) AS CD_MIC_PRFL_CLI,
    VARCHAR(TRIM(NM_MIC_PRFL_CLI), 150) AS NM_MIC_PRFL_CLI,
    BIGINT(COUNT(*)) AS QT_CLIENTES,
    MIN(DT_REF) AS MENOR_DT_REF_ESCOLHIDA,
    MAX(DT_REF) AS MAIOR_DT_REF_ESCOLHIDA
FROM ORDENADO
WHERE RN = 1
GROUP BY
    CD_MAC_PRFL_CLI,
    NM_MAC_PRFL_CLI,
    CD_MIC_PRFL_CLI,
    NM_MIC_PRFL_CLI
ORDER BY QT_CLIENTES DESC
"""

# ESTUDO 02 — GRUPO + CATEGORIA + NATUREZA REAL EM 2026
#
# Perguntas:
#   - Para cada GRUPO + CATEGORIA, o que o catálogo diz?
#   - Como a transação realmente apareceu em 2026: C / D / NULL / VAZIO / OUTRO?
#   - A mesma categoria apareceu como C e D?
#   - A natureza real diverge do catálogo?
#   - A categoria existe na transação mas não no catálogo?
#   - Como categorias 0 e 83 realmente chegam?
#   - Existem candidatos reais para alimentar a Classe Radar 4 (Crédito)?
#
# Estratégia:
#   A tabela transacional é bilionária.
#   O DB2 devolve somente agregados por PTC + categoria + natureza.
#   O join com os catálogos pequenos é feito depois no Spark.
####################################################################################################

%%spark

from pyspark.sql import functions as F
from pyspark import StorageLevel

DATA_INI = "2026-01-01"
DATA_FIM_EXCLUSIVA = "2027-01-01"

FAIXAS_PTC = [
    (1, 25, "G1"),
    (26, 50, "G2"),
    (51, 75, "G3"),
    (76, 100, "G4"),
]

print("\n" + "=" * 140)
print("ESTUDO 02 — GRUPO + CATEGORIA + NATUREZA REAL EM 2026")
print("=" * 140)


# --------------------------------------------------------------------------------------------------
# 1. Agrega a origem transacional diretamente no DB2.
# --------------------------------------------------------------------------------------------------

dfs = []

for ptc_min, ptc_max, rotulo in FAIXAS_PTC:

    sql = f"""
    SELECT
        SMALLINT(NR_PTC) AS NR_PTC,
        INTEGER(CD_CTGR_TRAN_OGNL) AS CD_CTGR_TRAN,
        CASE
            WHEN CD_NTZ_CTB_TRAN IS NULL THEN 'NULL'
            WHEN TRIM(CD_NTZ_CTB_TRAN) = '' THEN 'VAZIO'
            ELSE VARCHAR(TRIM(CD_NTZ_CTB_TRAN), 10)
        END AS NTZ_OBSERVADA,
        BIGINT(COUNT(*)) AS QT_TRANS,
        DECIMAL(SUM(ABS(COALESCE(VL_TRAN, 0))), 31, 2) AS VL_ABS
    FROM DB2GFP.TRAN_RLZD_INST_PCT
    WHERE NR_PTC BETWEEN {ptc_min} AND {ptc_max}
      AND CD_EST_TRAN_INST = 0
      AND CD_TIP_PSS = 1
      AND DT_TRAN >= DATE('{DATA_INI}')
      AND DT_TRAN < DATE('{DATA_FIM_EXCLUSIVA}')
    GROUP BY
        NR_PTC,
        CD_CTGR_TRAN_OGNL,
        CASE
            WHEN CD_NTZ_CTB_TRAN IS NULL THEN 'NULL'
            WHEN TRIM(CD_NTZ_CTB_TRAN) = '' THEN 'VAZIO'
            ELSE VARCHAR(TRIM(CD_NTZ_CTB_TRAN), 10)
        END
    """

    df_bloco = conector_db2.sql(
        sql,
        fetchsize=10_000,
        query_timeout=15 * 60,
        partition_column="NR_PTC",
        lower_bound=ptc_min,
        upper_bound=ptc_max + 1,
        num_partitions=4,
    )

    dfs.append(df_bloco)

df_trans = dfs[0]
for df_tmp in dfs[1:]:
    df_trans = df_trans.unionByName(df_tmp)

df_trans = (
    df_trans
    .groupBy("CD_CTGR_TRAN", "NTZ_OBSERVADA")
    .agg(
        F.sum("QT_TRANS").alias("QT_TRANS"),
        F.sum("VL_ABS").alias("VL_ABS"),
    )
    .persist(StorageLevel.MEMORY_AND_DISK)
)

df_trans.count()


# --------------------------------------------------------------------------------------------------
# 2. Catálogo oficial de categorias e grupos.
# --------------------------------------------------------------------------------------------------

df_cat = conector_db2.sql(
    """
    SELECT
        INTEGER(CD_CTGR_TRAN) AS CD_CTGR_TRAN,
        SMALLINT(CD_GR_CTGR_TRAN) AS CD_GR_CTGR_TRAN,
        CASE
            WHEN CD_NTZ_CTB_TRAN IS NULL THEN 'NULL'
            WHEN TRIM(CD_NTZ_CTB_TRAN) = '' THEN 'VAZIO'
            ELSE VARCHAR(TRIM(CD_NTZ_CTB_TRAN), 10)
        END AS NTZ_CATALOGO,
        VARCHAR(TRIM(TX_DCR_CTGR_TRAN), 150) AS TX_CATEGORIA
    FROM DB2GFP.CTGR_TRAN_OPB
    WHERE IN_CTGR_PDRO_SIS = 'S'
    """,
    fetchsize=10_000,
    query_timeout=15 * 60,
)

df_grupo = conector_db2.sql(
    """
    SELECT
        SMALLINT(CD_GR_CTGR_TRAN) AS CD_GR_CTGR_TRAN,
        VARCHAR(TRIM(TX_DCR_GR_CTGR), 120) AS TX_GRUPO
    FROM DB2GFP.GR_CTGR_TRAN
    """,
    fetchsize=10_000,
    query_timeout=15 * 60,
)

df_catalogo = (
    df_cat
    .join(df_grupo, "CD_GR_CTGR_TRAN", "left")
    .select(
        "CD_CTGR_TRAN",
        "CD_GR_CTGR_TRAN",
        "TX_GRUPO",
        "TX_CATEGORIA",
        "NTZ_CATALOGO",
    )
)


# --------------------------------------------------------------------------------------------------
# 3. Junta agregado real de 2026 ao catálogo.
# --------------------------------------------------------------------------------------------------

df_raw = (
    df_trans
    .join(df_catalogo, "CD_CTGR_TRAN", "left")
    .persist(StorageLevel.MEMORY_AND_DISK)
)

df_raw.count()

df_dominio = (
    df_raw
    .groupBy(
        "CD_GR_CTGR_TRAN",
        "TX_GRUPO",
        "CD_CTGR_TRAN",
        "TX_CATEGORIA",
        "NTZ_CATALOGO",
    )
    .agg(
        F.sum("QT_TRANS").alias("QT_TOTAL"),

        F.sum(
            F.when(F.col("NTZ_OBSERVADA") == "C", F.col("QT_TRANS"))
             .otherwise(0)
        ).alias("QT_C"),

        F.sum(
            F.when(F.col("NTZ_OBSERVADA") == "D", F.col("QT_TRANS"))
             .otherwise(0)
        ).alias("QT_D"),

        F.sum(
            F.when(F.col("NTZ_OBSERVADA") == "NULL", F.col("QT_TRANS"))
             .otherwise(0)
        ).alias("QT_NULL"),

        F.sum(
            F.when(F.col("NTZ_OBSERVADA") == "VAZIO", F.col("QT_TRANS"))
             .otherwise(0)
        ).alias("QT_VAZIO"),

        F.sum(
            F.when(
                ~F.col("NTZ_OBSERVADA").isin("C", "D", "NULL", "VAZIO"),
                F.col("QT_TRANS")
            ).otherwise(0)
        ).alias("QT_OUTRO"),

        F.sum("VL_ABS").alias("VL_ABS_TOTAL"),
        F.sort_array(F.collect_set("NTZ_OBSERVADA")).alias("NATUREZAS_OBSERVADAS"),
    )
    .withColumn(
        "STATUS",
        F.when(
            F.col("TX_CATEGORIA").isNull(),
            "SEM_CATALOGO"
        )
        .when(
            (F.col("QT_NULL") > 0)
            | (F.col("QT_VAZIO") > 0)
            | (F.col("QT_OUTRO") > 0),
            "NATUREZA_ANOMALA_NA_TRANSACAO"
        )
        .when(
            (F.col("QT_C") > 0) & (F.col("QT_D") > 0),
            "C_E_D_NA_MESMA_CATEGORIA"
        )
        .when(
            (F.col("NTZ_CATALOGO") == "C") & (F.col("QT_D") > 0),
            "DIVERGE_CATALOGO_C"
        )
        .when(
            (F.col("NTZ_CATALOGO") == "D") & (F.col("QT_C") > 0),
            "DIVERGE_CATALOGO_D"
        )
        .when(
            F.col("NTZ_CATALOGO").isin("NULL", "VAZIO")
            & ((F.col("QT_C") > 0) | (F.col("QT_D") > 0)),
            "CATALOGO_SEM_NATUREZA_ORIGEM_TEM"
        )
        .otherwise("OK")
    )
    .persist(StorageLevel.MEMORY_AND_DISK)
)

df_dominio.count()


# --------------------------------------------------------------------------------------------------
# 4. Prints para decisão.
# --------------------------------------------------------------------------------------------------

print("\n1. RESUMO DOS STATUS")
(
    df_dominio
    .groupBy("STATUS")
    .agg(
        F.count("*").alias("QT_CATEGORIAS"),
        F.sum("QT_TOTAL").alias("QT_TRANS"),
        F.sum("VL_ABS_TOTAL").alias("VL_ABS"),
    )
    .orderBy(F.desc("QT_TRANS"))
    .show(100, truncate=False)
)

print("\n2. DOMÍNIO COMPLETO — UMA LINHA POR GRUPO + CATEGORIA")
(
    df_dominio
    .select(
        "CD_GR_CTGR_TRAN",
        "TX_GRUPO",
        "CD_CTGR_TRAN",
        "TX_CATEGORIA",
        "NTZ_CATALOGO",
        "QT_C",
        "QT_D",
        "QT_NULL",
        "QT_VAZIO",
        "QT_OUTRO",
        "QT_TOTAL",
        "NATUREZAS_OBSERVADAS",
        "STATUS",
    )
    .orderBy(
        F.col("CD_GR_CTGR_TRAN").asc_nulls_last(),
        F.col("CD_CTGR_TRAN").asc_nulls_last(),
    )
    .show(500, truncate=False)
)

print("\n3. SOMENTE O QUE PRECISA SER INVESTIGADO")
(
    df_dominio
    .where(F.col("STATUS") != "OK")
    .select(
        "CD_GR_CTGR_TRAN",
        "TX_GRUPO",
        "CD_CTGR_TRAN",
        "TX_CATEGORIA",
        "NTZ_CATALOGO",
        "QT_C",
        "QT_D",
        "QT_NULL",
        "QT_VAZIO",
        "QT_OUTRO",
        "QT_TOTAL",
        "VL_ABS_TOTAL",
        "NATUREZAS_OBSERVADAS",
        "STATUS",
    )
    .orderBy(F.desc("QT_TOTAL"))
    .show(500, truncate=False)
)

print("\n4. FOCO RADAR 0 — CATEGORIAS 0 E 83")
(
    df_dominio
    .where(F.col("CD_CTGR_TRAN").isin(0, 83))
    .select(
        "CD_GR_CTGR_TRAN",

dfs_cpf = []

for ptc_min, ptc_max, rotulo in FAIXAS_PTC:

    sql = f"""

        WHERE DT_REF <= DATE('{DATA_PUBLICO_FIM}')
          AND TX_CTGR_MOD IN ('behaviour', 'bureau')
        GROUP BY TX_CTGR_MOD
        """
    )
    .collect()
)

map_part = {r["TX_CTGR_MOD"]: r["MAX_DT_REF"] for r in particoes}

if "behaviour" not in map_part or "bureau" not in map_part:
    raise ValueError(f"Partições RDPR incompletas: {map_part}")

DT_BEHAVIOUR = map_part["behaviour"]
DT_BUREAU = map_part["bureau"]

print("\n3. PARTIÇÕES USADAS PELA REGRA ATUAL")
print("behaviour:", DT_BEHAVIOUR)
print("bureau   :", DT_BUREAU)


# --------------------------------------------------------------------------------------------------
# 4. Avalia Behaviour e Bureau nas partições latest.
# --------------------------------------------------------------------------------------------------

df_latest = spark.sql(
    f"""
    SELECT
        CAST(NR_CPF AS DECIMAL(11,0)) AS NR_CPF,
        TX_CTGR_MOD AS MODELO,
        CAST(VL_RDPR AS DECIMAL(15,2)) AS VL_RDPR,
        CAST(CD_IN_UTZO AS SMALLINT) AS CD_IN_UTZO,
        DT_REF
    FROM HIVE_D1Q.RDPR_PF
    WHERE
        (
            (TX_CTGR_MOD = 'behaviour' AND DT_REF = DATE('{DT_BEHAVIOUR}'))
            OR
            (TX_CTGR_MOD = 'bureau' AND DT_REF = DATE('{DT_BUREAU}'))
        )
      AND CD_IN_UTZO = -1
      AND NR_CPF IS NOT NULL
    """
)

df_latest_av = (
    df_latest
    .groupBy("NR_CPF", "MODELO")
    .agg(
        F.count("*").alias("QT_REGISTROS"),
        F.max("VL_RDPR").alias("VL_RDPR_ORIGEM"),
        F.max("DT_REF").alias("DT_REF_MODELO"),
    )
    .withColumn(
        "CD_STS_MODELO",
        F.when(F.col("QT_REGISTROS") > 1, -5)
         .when(F.col("VL_RDPR_ORIGEM").isNull(), -6)
         .when(F.col("VL_RDPR_ORIGEM") < 0, -7)
         .otherwise(0)
         .cast("smallint")
    )
    .persist(StorageLevel.MEMORY_AND_DISK)
)

df_latest_av.count()

print("\n4. QUALIDADE DAS PARTIÇÕES LATEST")
(
    df_latest_av
    .groupBy("MODELO", "CD_STS_MODELO")
    .agg(
        F.count("*").alias("QT_CPFS"),
        F.min("DT_REF_MODELO").alias("DT_REF"),
    )
    .orderBy("MODELO", "CD_STS_MODELO")
    .show(100, truncate=False)
)


# --------------------------------------------------------------------------------------------------
# 5. Reproduz a prioridade Behaviour > Bureau e compara com a ANA.
# --------------------------------------------------------------------------------------------------

bh = (
    df_latest_av
    .where(F.col("MODELO") == "behaviour")
    .select(
        "NR_CPF",
        F.col("CD_STS_MODELO").alias("STS_BH"),
        F.col("VL_RDPR_ORIGEM").alias("VL_BH"),
        F.col("DT_REF_MODELO").alias("DT_BH"),
    )
)

bu = (
    df_latest_av
    .where(F.col("MODELO") == "bureau")
    .select(
        "NR_CPF",
        F.col("CD_STS_MODELO").alias("STS_BU"),
        F.col("VL_RDPR_ORIGEM").alias("VL_BU"),
        F.col("DT_REF_MODELO").alias("DT_BU"),
    )
)

df_recalc = (
    df_ana
    .join(df_cpf, "CD_CLI", "left")
    .join(bh, "NR_CPF", "left")
    .join(bu, "NR_CPF", "left")
    .withColumn(
        "MODELO_ESCOLHIDO",
        F.when(F.col("STS_BH").isNotNull(), "behaviour")
         .when(F.col("STS_BU").isNotNull(), "bureau")
    )
    .withColumn(
        "STS_MODELO_ESCOLHIDO",
        F.when(F.col("STS_BH").isNotNull(), F.col("STS_BH"))
         .when(F.col("STS_BU").isNotNull(), F.col("STS_BU"))
         .otherwise(F.lit(-1))
    )
    .withColumn(
        "VL_MODELO_ESCOLHIDO",
        F.when(F.col("STS_BH").isNotNull(), F.col("VL_BH"))
         .when(F.col("STS_BU").isNotNull(), F.col("VL_BU"))
    )
    .withColumn(
        "VL_REN_PRES_ESPERADO",
        F.when(
            F.col("CD_STS_CPF_RECALC") < 0,
            F.col("CD_STS_CPF_RECALC").cast("decimal(18,2)")
        )
        .when(
            F.col("STS_MODELO_ESCOLHIDO") == 0,
            F.col("VL_MODELO_ESCOLHIDO").cast("decimal(18,2)")
        )
        .otherwise(
            F.col("STS_MODELO_ESCOLHIDO").cast("decimal(18,2)")
        )
    )
    .persist(StorageLevel.MEMORY_AND_DISK)
)

df_recalc.count()

print("\n5. CAUSA DIRETA PELA REGRA ATUAL")
(
    df_recalc
    .withColumn(
        "CAUSA_ATUAL",
        F.when(F.col("CD_STS_CPF_RECALC") == -4, "CPF_AMBIGUO_NA_TABELA_PRINCIPAL")
         .when(F.col("CD_STS_CPF_RECALC") == -2, "CPF_AUSENTE_OU_INVALIDO_NA_TABELA_PRINCIPAL")
         .when(
             (F.col("CD_STS_CPF_RECALC") == 0)
             & (F.col("STS_MODELO_ESCOLHIDO") == -1),
             "CPF_VALIDO_SEM_MODELO_NAS_PARTICOES_LATEST"
         )
         .when(F.col("STS_MODELO_ESCOLHIDO") == -5, "MODELO_ESCOLHIDO_DUPLICADO")
         .when(F.col("STS_MODELO_ESCOLHIDO") == -6, "MODELO_ESCOLHIDO_VL_RDPR_NULL")
         .when(F.col("STS_MODELO_ESCOLHIDO") == -7, "MODELO_ESCOLHIDO_VL_RDPR_NEGATIVO")
         .when(F.col("STS_MODELO_ESCOLHIDO") == 0, "MODELO_VALIDO")
         .otherwise("OUTRO")
    )
    .groupBy("CAUSA_ATUAL", "MODELO_ESCOLHIDO")
    .agg(F.count("*").alias("QT_CLIENTES"))
    .orderBy(F.desc("QT_CLIENTES"))
    .show(100, truncate=False)
)

print("\n5.1 DIVERGÊNCIAS — ANA x REGRA RECALCULADA")
(
    df_recalc
    .where(~F.col("VL_REN_PRES").eqNullSafe(F.col("VL_REN_PRES_ESPERADO")))
    .select(
        "CD_CLI",
        "VL_REN_PRES",
        "VL_REN_PRES_ESPERADO",
        "CD_STS_CPF_RECALC",
        "MODELO_ESCOLHIDO",
        "STS_MODELO_ESCOLHIDO",
    )
    .show(100, truncate=False)
)


# --------------------------------------------------------------------------------------------------
# 6. Abre especificamente o -1:
#    - existe em partição anterior?
#    - existe RDPR mas CD_IN_UTZO diferente de -1?
#    - não existe registro para o CPF?
# --------------------------------------------------------------------------------------------------

df_menos1 = (
    df_recalc
    .where(
        (F.col("VL_REN_PRES") == -1)
        & (F.col("CD_STS_CPF_RECALC") == 0)
        & F.col("NR_CPF").isNotNull()
    )
    .select("CD_CLI", "NR_CPF")
    .persist(StorageLevel.MEMORY_AND_DISK)
)

df_menos1.count()

df_cpfs_menos1 = df_menos1.select("NR_CPF").distinct()

df_hist = (
    spark.table("HIVE_D1Q.RDPR_PF")
    .where(
        (F.col("DT_REF") >= F.to_date(F.lit(DATA_HIST_INI_RDPR)))
        & (F.col("DT_REF") <= F.to_date(F.lit(str(DATA_PUBLICO_FIM))))
        & F.col("TX_CTGR_MOD").isin("behaviour", "bureau")
        & F.col("NR_CPF").isNotNull()
    )
    .select(
        F.col("NR_CPF").cast("decimal(11,0)").alias("NR_CPF"),
        "TX_CTGR_MOD",
        "DT_REF",
        "CD_IN_UTZO",
        "VL_RDPR",
    )
    .join(df_cpfs_menos1, "NR_CPF", "left_semi")
)

df_hist_resumo = (
    df_hist
    .groupBy("NR_CPF")
    .agg(
        F.count("*").alias("QT_RDPR_HIST"),
        F.max(
            F.when(F.col("CD_IN_UTZO") == -1, 1).otherwise(0)
        ).alias("TEM_HIST_ELEGIVEL"),
        F.max(
            F.when(F.col("CD_IN_UTZO") != -1, 1).otherwise(0)
        ).alias("TEM_HIST_OUTRO_CD_IN_UTZO"),
        F.max(
            F.when(F.col("CD_IN_UTZO") == -1, F.col("DT_REF"))
        ).alias("ULT_DT_REF_ELEGIVEL"),
        F.sort_array(
            F.collect_set(
                F.when(F.col("CD_IN_UTZO") == -1, F.col("TX_CTGR_MOD"))
            )
        ).alias("MODELOS_HIST_ELEGIVEIS"),
    )
)

df_causa_menos1 = (
    df_menos1
    .join(df_hist_resumo, "NR_CPF", "left")
    .fillna(
        {
            "QT_RDPR_HIST": 0,
            "TEM_HIST_ELEGIVEL": 0,
            "TEM_HIST_OUTRO_CD_IN_UTZO": 0,
        }
    )
    .withColumn(
        "CAUSA_MENOS_1",
        F.when(
            F.col("TEM_HIST_ELEGIVEL") == 1,
            "EXISTE_MODELO_EM_PARTICAO_ANTERIOR_MAS_NAO_NA_LATEST"
        )
        .when(
            (F.col("QT_RDPR_HIST") > 0)
            & (F.col("TEM_HIST_OUTRO_CD_IN_UTZO") == 1),
            "RDPR_EXISTE_MAS_CD_IN_UTZO_NAO_E_MENOS_1"
        )
        .when(
            F.col("QT_RDPR_HIST") == 0,
            "CPF_VALIDO_SEM_REGISTRO_RDPR_DESDE_2025"
        )
        .otherwise("OUTRA_CAUSA")
    )
    .persist(StorageLevel.MEMORY_AND_DISK)
)

df_causa_menos1.count()

print("\n6. CAUSA RAIZ DOS CLIENTES QUE TERMINARAM EM -1")
(
    df_causa_menos1
    .groupBy("CAUSA_MENOS_1")
    .agg(
        F.count("*").alias("QT_CLIENTES"),
        F.min("ULT_DT_REF_ELEGIVEL").alias("MENOR_ULT_DT_REF"),
        F.max("ULT_DT_REF_ELEGIVEL").alias("MAIOR_ULT_DT_REF"),
    )
    .orderBy(F.desc("QT_CLIENTES"))
    .show(100, truncate=False)
)

print("\n6.1 AMOSTRAS POR CAUSA DO -1")
causas = [r["CAUSA_MENOS_1"] for r in df_causa_menos1.select("CAUSA_MENOS_1").distinct().collect()]

for causa in sorted(causas):
    print("\n---", causa)
    (
        df_causa_menos1
        .where(F.col("CAUSA_MENOS_1") == causa)
        .select(
            "CD_CLI",
            "NR_CPF",
            "QT_RDPR_HIST",
            "TEM_HIST_ELEGIVEL",
            "TEM_HIST_OUTRO_CD_IN_UTZO",
            "ULT_DT_REF_ELEGIVEL",
            "MODELOS_HIST_ELEGIVEIS",
        )
        .limit(20)
        .show(20, truncate=False)
    )
